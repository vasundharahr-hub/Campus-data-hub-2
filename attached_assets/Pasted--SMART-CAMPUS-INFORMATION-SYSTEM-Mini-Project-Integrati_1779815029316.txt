import os
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# -------------------------------
# STUDENT REGISTRATION MODULE
# -------------------------------

students = []

def register_student():
    roll = input("Enter Roll Number: ")
    name = input("Enter Student Name: ")
    dept = input("Enter Department: ")

    marks = []
    for i in range(3):
        mark = int(input(f"Enter Subject {i+1} Marks: "))
        marks.append(mark)

    avg = sum(marks)/len(marks)

    if avg >= 90:
        grade = "A"
    elif avg >= 75:
        grade = "B"
    elif avg >= 60:
        grade = "C"
    else:
        grade = "D"

    student = {
        "Roll": roll,
        "Name": name,
        "Department": dept,
        "Marks": marks,
        "Average": avg,
        "Grade": grade
    }

    students.append(student)

    print("\nStudent Registered Successfully")
    print("Grade:", grade)


# -------------------------------
# COURSE ENROLLMENT MODULE
# -------------------------------

def enroll_course():
    roll = input("Enter Roll Number: ")
    course = input("Enter Course Name: ")

    for student in students:
        if student["Roll"] == roll:
            student["Course"] = course
            print("Course Enrolled Successfully")
            return

    print("Student not found")


# -------------------------------
# DISPLAY STUDENT RECORDS
# -------------------------------

def display_students():

    if len(students) == 0:
        print("No records found")
        return

    for s in students:
        print("\n----------------")
        print("Roll:", s["Roll"])
        print("Name:", s["Name"])
        print("Department:", s["Department"])
        print("Marks:", s["Marks"])
        print("Average:", s["Average"])
        print("Grade:", s["Grade"])
        print("Course:", s.get("Course", "Not Enrolled"))


# -------------------------------
# SEARCH STUDENT
# -------------------------------

def search_student():

    roll = input("Enter Roll Number to search: ")

    for s in students:
        if s["Roll"] == roll:
            print("Student Found")
            print(s)
            return

    print("Student not found")


# -------------------------------
# SORT STUDENTS
# -------------------------------

def sort_students():

    sorted_list = sorted(
        students,
        key=lambda x: x["Average"],
        reverse=True
    )

    print("\nSorted Student List")

    for s in sorted_list:
        print(s["Name"], "-", s["Average"])


# -------------------------------
# FEE CALCULATION MODULE
# -------------------------------

def calculate_fee():

    tuition = float(input("Enter Tuition Fee: "))
    lab = float(input("Enter Lab Fee: "))
    exam = float(input("Enter Exam Fee: "))

    total = tuition + lab + exam

    print("Total Fee =", total)


# -------------------------------
# FILE MANAGEMENT
# -------------------------------

def save_records():

    file = open("academic_records.txt", "w")

    for s in students:
        file.write(
            f'{s["Roll"]},'
            f'{s["Name"]},'
            f'{s["Department"]},'
            f'{s["Average"]},'
            f'{s["Grade"]}\n'
        )

    file.close()

    print("Records Saved")


def load_records():

    try:

        file = open(
            "academic_records.txt",
            "r"
        )

        print("\nAcademic Records")

        print(file.read())

        file.close()

    except FileNotFoundError:
        print("Record file not found")


# -------------------------------
# DIRECTORY SCANNING
# -------------------------------

def scan_directory():

    path = input(
        "Enter directory path: "
    )

    try:

        files = os.listdir(path)

        print("\nFiles Present")

        for f in files:
            print(f)

    except Exception as e:
        print("Error:", e)


# -------------------------------
# ANALYTICS USING NUMPY/PANDAS
# -------------------------------

def performance_analytics():

    if len(students) == 0:
        print("No data available")
        return

    names = []
    averages = []

    for s in students:
        names.append(s["Name"])
        averages.append(s["Average"])

    arr = np.array(averages)

    print("\nAnalytics")

    print(
        "Mean Marks:",
        np.mean(arr)
    )

    print(
        "Maximum:",
        np.max(arr)
    )

    print(
        "Minimum:",
        np.min(arr)
    )

    df = pd.DataFrame({
        "Student": names,
        "Average": averages
    })

    print("\nPandas DataFrame")

    print(df)

    plt.bar(names, averages)

    plt.xlabel("Students")

    plt.ylabel("Average Marks")

    plt.title(
        "Student Performance Analysis"
    )

    plt.show()


# -------------------------------
# MAIN MENU
# -------------------------------

while True:

    print("\nSMART CAMPUS SYSTEM")

    print("1.Register Student")
    print("2.Course Enrollment")
    print("3.Display Records")
    print("4.Search Student")
    print("5.Sort Students")
    print("6.Fee Calculation")
    print("7.Save Records")
    print("8.Load Records")
    print("9.Directory Scan")
    print("10.Analytics")
    print("11.Exit")

    choice = int(
        input("Enter choice: ")
    )

    if choice == 1:
        register_student()

    elif choice == 2:
        enroll_course()

    elif choice == 3:
        display_students()

    elif choice == 4:
        search_student()

    elif choice == 5:
        sort_students()

    elif choice == 6:
        calculate_fee()

    elif choice == 7:
        save_records()

    elif choice == 8:
        load_records()

    elif choice == 9:
        scan_directory()

    elif choice == 10:
        performance_analytics()

    elif choice == 11:
        print("Program Ended")
        break

    else:
        print("Invalid Choice")
